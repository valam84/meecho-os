# Веха 10.1: подъём xHCI на плате. Загрузить MEECHO и держать окно, пока
# драйвер ставится по ssh и запускается вручную.
#
# Почему вручную, а не из /etc/rc: корень на eMMC уже собран, и заменять его
# образом ради одной программы дорого - ssh кладёт файл в живую систему за
# секунды. В rc драйвер попадёт, когда будет что запускать (10.4).
#
# Всё, что печатает драйвер, идёт на консоль, а не в ssh: у MEECHO нет
# syslog. Поэтому окно держится с открытой консолью, а не закрывается сразу
# после входа.
120 WAIT switching console to 115200
1 BAUD 115200
2 <CR>
1 <CR>
420 WAIT login:
2 root
60 WAIT they are.
2 export TERM=vt100 PAGER=cat
3 uname -s; cat /proc/uptime; ifconfig | grep 'inet '; printf 'd%s\n' 1
40 WAIT d1
# Окно: с рабочей машины кладём /service/xhci и поднимаем его.
300 WAIT ---hold-open---
3 printf 'd%s\n' 2
40 WAIT d2
2 /sbin/reboot
20
