# Замер скорости USB после починки: загрузить MEECHO и держать окно.
#
# Консоль здесь нужна только чтобы довести систему до входа и оставить её
# живой; драйверы ставятся и гоняются по ssh с рабочей машины. Окно
# длинное, поэтому запускать с board-cycle.sh -l - сторож на час.
#
# Метки конца команды - printf, а не echo: терминал отражает набранное, и
# echo нашёлся бы в журнале раньше, чем команда выполнилась.
120 WAIT switching console to 115200
1 BAUD 115200
2 <CR>
1 <CR>
420 WAIT login:
2 root
60 WAIT they are.
2 export TERM=vt100 PAGER=cat
3 uname -s; cat /proc/uptime; ifconfig | grep 'inet '; printf 'd%s\n' 1
40 WAIT d1
# Окно: ставим драйверы и меряем по ssh.
1800 WAIT ---hold-open---
3 cat /proc/uptime; printf 'd%s\n' 2
40 WAIT d2
2 /sbin/reboot
20
