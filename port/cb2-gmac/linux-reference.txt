############ 7. Питание PHY

# phy-supply = SWITCH_REG1 у PMIC:
# 				SWITCH_REG1 {
# 					regulator-boot-on;
# 					regulator-always-on;
# 					regulator-name = "vcc_3v3";
# 					phandle = <0x30>;
# 
# 					regulator-state-mem {
# 						regulator-off-in-suspend;
# 					};
# regulator-always-on и regulator-boot-on: включается загрузчиком и
# никогда не гасится, значит драйвера PMIC для сети не нужно.
